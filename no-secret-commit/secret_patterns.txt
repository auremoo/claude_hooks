# =============================================================================
# Secret Patterns - Scanned before git commit
# =============================================================================
# One regex pattern per line. Lines starting with # are comments.
# These patterns are matched against staged file contents.
# =============================================================================

# --- API Keys ---
(api[_-]?key|apikey)\s*[:=]\s*['"][a-zA-Z0-9]{16,}
(api[_-]?secret|apisecret)\s*[:=]\s*['"][a-zA-Z0-9]{16,}

# --- AWS ---
AKIA[0-9A-Z]{16}
aws[_-]?secret[_-]?access[_-]?key\s*[:=]

# --- Azure ---
AccountKey=[a-zA-Z0-9+/=]{40,}

# --- Google ---
AIza[0-9A-Za-z_-]{35}

# --- GitHub ---
gh[pousr]_[A-Za-z0-9_]{36,}
github_pat_[A-Za-z0-9_]{22,}

# --- Generic tokens ---
(access[_-]?token|auth[_-]?token|bearer)\s*[:=]\s*['"][a-zA-Z0-9._-]{20,}

# --- Private keys ---
-----BEGIN (RSA |EC |DSA |OPENSSH )?PRIVATE KEY-----
-----BEGIN PGP PRIVATE KEY BLOCK-----

# --- Passwords in config ---
(password|passwd|pwd)\s*[:=]\s*['"][^'"]{8,}
(DB_PASSWORD|DATABASE_PASSWORD|MYSQL_PASSWORD|POSTGRES_PASSWORD)\s*=

# --- Connection strings ---
(mongodb(\+srv)?|mysql|postgresql|redis|amqp):\/\/[^:]+:[^@]+@

# --- JWT / Secrets ---
(jwt[_-]?secret|secret[_-]?key|signing[_-]?key)\s*[:=]\s*['"][a-zA-Z0-9]{16,}

# --- Stripe ---
sk_(live|test)_[a-zA-Z0-9]{24,}
rk_(live|test)_[a-zA-Z0-9]{24,}

# --- Slack ---
xox[baprs]-[a-zA-Z0-9-]{10,}

# --- SendGrid ---
SG\.[a-zA-Z0-9_-]{22}\.[a-zA-Z0-9_-]{43}

# --- Twilio ---
SK[a-f0-9]{32}

# --- .env files with secrets ---
^[A-Z_]*(SECRET|TOKEN|KEY|PASSWORD|CREDENTIAL)[A-Z_]*\s*=\s*[^[:space:]]{8,}
