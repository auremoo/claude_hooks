# =============================================================================
# Naming Convention Rules
# =============================================================================
# Format: FILE_PATTERN | CHECK_TYPE | DETAILS
#
# FILE_PATTERN: glob-like regex for file types this rule applies to
# CHECK_TYPE: one of: class_prefix, interface_prefix, function_case, variable_case, file_case
# DETAILS: expected convention or prefix
#
# Cases: camelCase, PascalCase, snake_case, UPPER_SNAKE_CASE, kebab-case
# Several accepted cases can be combined with "/": camelCase/PascalCase
# =============================================================================

# --- TypeScript/JavaScript ---
\.(ts|tsx|js|jsx)$ | interface_prefix | I
\.(ts|tsx|js|jsx)$ | function_case | camelCase
\.(ts|tsx|js|jsx)$ | class_prefix |
\.(ts|tsx|js|jsx)$ | constant_case | UPPER_SNAKE_CASE

# --- Python ---
\.py$ | function_case | snake_case
\.py$ | class_case | PascalCase
\.py$ | constant_case | UPPER_SNAKE_CASE

# --- C# ---
\.cs$ | interface_prefix | I
\.cs$ | function_case | PascalCase
\.cs$ | variable_case | camelCase

# --- Go ---
\.go$ | function_case | camelCase/PascalCase
\.go$ | variable_case | camelCase
